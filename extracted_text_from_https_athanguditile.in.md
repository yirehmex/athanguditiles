        function toggleSelection(element) {
            // Find if already selected to deselect, or select if not
            if(element.classList.contains('pattern-selected')) {
                element.classList.remove('pattern-selected');
                element.querySelector('.material-symbols-outlined')?.remove();
            } else {
                element.classList.add('pattern-selected');
                // Optional: Add a checkmark overlay
                if(!element.querySelector('.check-mark')) {
                    const check = document.createElement('div');
                    check.className = "check-mark absolute top-3 right-3 bg-heritage-crimson text-white w-6 h-6 rounded-full flex items-center justify-center shadow-lg";
                    check.innerHTML = '<span class="material-symbols-outlined text-[16px]">check</span>';
                    element.querySelector('.aspect-square').appendChild(check);
                }
            }
        }

        function requestSuccess() {
            // Simple visual feedback
            const btn = event.currentTarget;
            const originalContent = btn.innerHTML;
            btn.innerHTML = '<span class="material-symbols-outlined animate-spin">sync</span> PROCESSING...';
            btn.disabled = true;
            
            setTimeout(() => {
                btn.innerHTML = '<span class="material-symbols-outlined">task_alt</span> KIT REQUESTED';
                btn.classList.replace('bg-heritage-crimson', 'bg-green-700');
                
                // Show a brief toast
                const toast = document.createElement('div');
                toast.className = "fixed top-10 left-1/2 -translate-x-1/2 bg-on-background text-white px-6 py-3 rounded-full font-label-sm text-sm z-[100] shadow-2xl transition-all opacity-0 translate-y-4";
                toast.innerText = "Sample request submitted successfully.";
                document.body.appendChild(toast);
                
                setTimeout(() => {
                    toast.classList.replace('opacity-0', 'opacity-100');
                    toast.classList.replace('translate-y-4', 'translate-y-0');
                }, 10);
            }, 1500);
        }
